# Terço App — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a Next.js 14 PWA for praying the Catholic rosary offline, with manual prayer advancement, progress saving in localStorage, and adjustable font size.

**Architecture:** Single Next.js page with client-side state machine controlling three views (selection → praying → complete). All prayer content is static TypeScript data generated by helper functions. Progress and font preference persist in localStorage. A PWA Service Worker caches all assets for full offline use after first visit.

**Tech Stack:** Next.js 14 (App Router, TypeScript), Tailwind CSS, `@ducanh2912/next-pwa` (Workbox), Google Fonts (Lora + Inter), Jest + Testing Library

function gerarEtapas(subMisterios: SubMisterio[]): Etapa[] {
  const etapas: Etapa[] = []

  etapas.push({ tipo: 'credo', label: 'Credo', texto: CREDO })
  etapas.push({ tipo: 'pai-nosso', label: 'Pai Nosso', texto: PAI_NOSSO })
  for (let i = 1; i <= 3; i++) {
    etapas.push({ tipo: 'ave-maria', label: `Ave Maria ${i}/3`, texto: AVE_MARIA })
  }
  etapas.push({ tipo: 'gloria', label: 'Glória', texto: GLORIA })

  subMisterios.forEach(({ numero, titulo, anuncio }) => {
    etapas.push({
      tipo: 'anuncio',
      label: `${numero}º Mistério — ${titulo}`,
      texto: anuncio,
    })
    etapas.push({ tipo: 'pai-nosso', label: 'Pai Nosso', texto: PAI_NOSSO })
    for (let i = 1; i <= 10; i++) {
      etapas.push({ tipo: 'ave-maria', label: `Ave Maria ${i}/10`, texto: AVE_MARIA })
    }
    etapas.push({ tipo: 'gloria', label: 'Glória', texto: GLORIA })
    etapas.push({ tipo: 'fatima', label: 'Oração de Fátima', texto: FATIMA })
  })

  etapas.push({ tipo: 'salve-rainha', label: 'Salve Rainha', texto: SALVE_RAINHA })

  return etapas
}

const PROGRESS_KEY = 'terco-progress'
const PREFS_KEY = 'terco-preferences'

const DEFAULT_PREFERENCES: Preferences = { fontSize: 'normal' }

export function saveProgress(progress: Progress): void {
  localStorage.setItem(PROGRESS_KEY, JSON.stringify(progress))
}

export function loadProgress(): Progress | null {
  const raw = localStorage.getItem(PROGRESS_KEY)
  if (!raw) return null
  try {
    return JSON.parse(raw) as Progress
  } catch {
    return null
  }
}

export function clearProgress(): void {
  localStorage.removeItem(PROGRESS_KEY)
}

export function savePreferences(prefs: Preferences): void {
  localStorage.setItem(PREFS_KEY, JSON.stringify(prefs))
}

export function loadPreferences(): Preferences {
  const raw = localStorage.getItem(PREFS_KEY)
  if (!raw) return DEFAULT_PREFERENCES
  try {
    return JSON.parse(raw) as Preferences
  } catch {
    return DEFAULT_PREFERENCES
  }
}

```tsx
'use client'

import { useState, useEffect, useCallback } from 'react'
import type { Misterio, FontSize } from '@/data/types'
import { canGoNext, canGoPrev, nextIndex, prevIndex } from '@/lib/navigation'
import { FontControls } from './FontControls'

const FONT_CLASS: Record<FontSize, string> = {
  normal: 'text-lg',
  grande: 'text-2xl',
  'muito-grande': 'text-3xl',
}

type Props = {
  misterio: Misterio
  initialIndex: number
  fontSize: FontSize
  onFontChange: (size: FontSize) => void
  onComplete: () => void
  onStepChange: (index: number) => void
}

export function PrayerScreen({
  misterio,
  initialIndex,
  fontSize,
  onFontChange,
  onComplete,
  onStepChange,
}: Props) {
  const [index, setIndex] = useState(initialIndex)
  const [visible, setVisible] = useState(true)

  const etapa = misterio.etapas[index]
  const total = misterio.etapas.length

  const go = useCallback(
    (newIndex: number) => {
      setVisible(false)
      setTimeout(() => {
        setIndex(newIndex)
        onStepChange(newIndex)
        setVisible(true)
      }, 150)
    },
    [onStepChange]
  )

  function handleNext() {
    if (canGoNext(index, total)) {
      go(nextIndex(index))
    } else {
      onComplete()
    }
  }

  function handlePrev() {
    if (canGoPrev(index)) go(prevIndex(index))
  }

  useEffect(() => {
    function onKey(e: KeyboardEvent) {
      if (e.key === 'ArrowRight' || e.key === ' ') {
        e.preventDefault()
        handleNext()
      }
      if (e.key === 'ArrowLeft') {
        e.preventDefault()
        handlePrev()
      }
    }
    window.addEventListener('keydown', onKey)
    return () => window.removeEventListener('keydown', onKey)
  })

  return (
    <div className="min-h-screen bg-[#FAF9F6] flex flex-col max-w-[680px] mx-auto px-6">
      {/* Header */}
      <div className="py-5 border-b border-[#E5E7EB]">
        <p className="text-xs text-[#6B7280] uppercase tracking-widest">
          {misterio.titulo}
        </p>
        <p className="text-sm text-[#8B7355] font-medium mt-0.5">{etapa.label}</p>
        <div className="mt-2 h-1 bg-[#E5E7EB] rounded-full overflow-hidden">
          <div
            className="h-full bg-[#8B7355] rounded-full transition-all duration-300"
            style={{ width: `${((index + 1) / total) * 100}%` }}
          />
        </div>
      </div>

      {/* Prayer text */}
      <div className="flex-1 flex items-center py-8 pr-14">
        <p
          className={`font-serif text-[#1C1C1E] leading-relaxed whitespace-pre-line transition-opacity duration-150 ${FONT_CLASS[fontSize]} ${visible ? 'opacity-100' : 'opacity-0'}`}
        >
          {etapa.texto}
        </p>
      </div>

      {/* Navigation */}
      <div className="py-6 flex gap-3">
        <button
          onClick={handlePrev}
          disabled={!canGoPrev(index)}
          className="px-5 py-4 rounded-lg border border-[#E5E7EB] text-[#6B7280] font-medium disabled:opacity-30 hover:border-[#8B7355] transition-colors"
        >
          ← Anterior
        </button>
        <button
          onClick={handleNext}
          className="flex-1 py-4 rounded-lg bg-[#8B7355] text-white font-medium text-lg hover:bg-[#7a6347] transition-colors"
        >
          {canGoNext(index, total) ? 'Próximo →' : 'Concluir ✓'}
        </button>
      </div>

      <FontControls fontSize={fontSize} onChange={onFontChange} />
    </div>
  )
}

    setView('praying')
  }

  function handleResume() {
    if (!savedProgress) return
    setTipo(savedProgress.tipo)
    setMisterioId(savedProgress.misterioId)
    setEtapaIndex(savedProgress.etapaIndex)
    setSavedProgress(null)
    setView('praying')
  }

  function handleDismissProgress() {
    clearProgress()
    setSavedProgress(null)
  }

  function handleStepChange(index: number) {
    setEtapaIndex(index)
    saveProgress({ tipo, misterioId, etapaIndex: index })
  }

  function handleComplete() {
    clearProgress()
    setSavedProgress(null)
    setView('complete')
  }

  function handleRestart() {
    setEtapaIndex(0)
    setView('selection')
  }

  function handleFontChange(size: FontSize) {
    setFontSize(size)
    savePreferences({ fontSize: size })
  }

  if (!hydrated) return null

  const misterio = TERCOS[tipo].misterios.find((m) => m.id === misterioId)

  return (
    <>
      {view === 'selection' && (
        <SelectionScreen
          savedProgress={savedProgress}
          onSelectMisterio={handleSelectMisterio}
          onResume={handleResume}
          onDismissProgress={handleDismissProgress}
        />
      )}
      {view === 'praying' && misterio && (
        <PrayerScreen
          misterio={misterio}
          initialIndex={etapaIndex}
          fontSize={fontSize}
          onFontChange={handleFontChange}
          onComplete={handleComplete}
          onStepChange={handleStepChange}
        />
      )}
      {view === 'complete' && <CompletionScreen onRestart={handleRestart} />}
    </>
  )
}
```
